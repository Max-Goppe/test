//@version=6
// RVP SCALP - multitimeframe market-structure indicator for XAUUSD (tuned on MT5 GOLD M5 export 2022-2026).
// Trend line + liquidity zones (BSL/SSL) + MTF confluence table (phase, TF trends, impulse, session color). No signals.
indicator("RVP SCALP", "RVP SCALP", overlay = true, max_labels_count = 100, max_boxes_count = 50, max_lines_count = 100, precision = 2)

// --- Auto presets: settings follow the CHART timeframe (timeframe.period), not an input.
// Tuned on MT5 GOLD M5 export 2022-2026 (entry TF = chart TF; confirm = x3; trend = x9..x12).
tfSec = timeframe.in_seconds(timeframe.period)
isM1 = tfSec <= 60
isM5 = tfSec > 60 and tfSec <= 300
isM15 = tfSec > 300 and tfSec <= 900
isUp = tfSec > 900
// Scalping focus: presets tuned for M1/M5/M15 only (backtest on MT5 GOLD 2022-2026).
// Above M15 the scalping edge disappears, so higher charts fall back to the M15 profile.
cfgRegmeLen = isM15 or isUp ? 55 : isM5 ? 34 : 24
cfgVolMult = isM15 or isUp ? 2.0 : isM5 ? 2.0 : 1.8
cfgMinBodyAtr = isM15 or isUp ? 0.50 : isM5 ? 0.50 : 0.85

groupTf = "Timeframes (scalping: M1-M15)"
entryTfIn   = input.timeframe("", "Entry timeframe (empty = chart)", group = groupTf)
confirmTfIn = input.timeframe("", "Confirm timeframe (empty = auto x3 of chart)", group = groupTf)
trendTfIn   = input.timeframe("", "Trend timeframe (empty = auto x9-x12 of chart)", group = groupTf)
// Auto MTF ladder follows the CHART timeframe: confirm ~x3, trend ~x9-x12.
// Capped at M15: on charts above M15 the ladder stays at 15/60/240.
cfgConfirmTf = tfSec <= 60 ? "3" : tfSec <= 300 ? "15" : "60"
cfgTrendTf   = tfSec <= 60 ? "5" : tfSec <= 300 ? "60" : "240"
entryTf   = entryTfIn
confirmTf = confirmTfIn != "" ? confirmTfIn : cfgConfirmTf
trendTf   = trendTfIn != "" ? trendTfIn : cfgTrendTf

groupCore = "REGME core (per TF)"
regmeLengthIn = input.int(0, "REGME length (0 = auto by chart TF)", minval = 0, group = groupCore, tooltip = "Auto: M1=24, M5=34, M15=55")
regmeSmooth   = input.int(8, "REGME smoothing", minval = 1, group = groupCore)
atrLength     = input.int(14, "ATR length", minval = 1, group = groupCore)
volumeBaseLen = input.int(50, "Volume baseline", minval = 5, group = groupCore)
volMultEntryIn = input.float(0.0, "Vol multiplier entry TF (0 = auto)", minval = 0.0, step = 0.1, group = groupCore, tooltip = "Auto: M5/M15=2.0, M1=1.8. Backtest: vol x1.3 on M5 is pure noise, keep >= 1.6.")
volMultBig    = input.float(2.0, "Vol multiplier (confirm/trend TF)", minval = 0.5, step = 0.1, group = groupCore)
minBodyAtrIn = input.float(0.0, "Min body / ATR (0 = auto)", minval = 0.0, step = 0.05, group = groupCore, tooltip = "Auto: M5/M15=0.50, M1=0.85")


groupSignal = "VDO (объёмное давление)"
vdoLength = input.int(14, "VDO length", minval = 2, group = groupSignal)
vdoSigLen = input.int(8, "VDO signal length", minval = 1, group = groupSignal)

groupSession = "Торговые окна (время Екатеринбурга, UTC+5)"
useSessionFilter = input.bool(true, "Подсвечивать торговое окно в таблице", group = groupSession, tooltip = "Строка «Торговое окно» в таблице: зелёный — торгуем (Европа 11-15, США 17-21 ЕКБ), красный — вне окон. Backtest GOLD M5 2022-2026: эти два окна дают лучшее качество движения; азия/ночь - худшее.")
sessEurope = input.bool(true, "Окно 1: Европа 11:00-15:00 (ЕКБ)", group = groupSession)
sessUs = input.bool(true, "Окно 2: США 17:00-21:00 (ЕКБ)", group = groupSession)
sessCustomFrom = input.int(-1, "Своё окно: час от (-1 = выкл, время ЕКБ)", minval = -1, maxval = 23, group = groupSession)
sessCustomTo = input.int(23, "Своё окно: час до (время ЕКБ)", minval = 0, maxval = 23, group = groupSession)

groupVisuals = "Visuals"
showTable = input.bool(true, "Show status table", group = groupVisuals)
tableCorner = input.string("Верх справа", "Позиция таблицы", options = ["Верх слева", "Верх справа", "Низ слева", "Низ справа"], group = groupVisuals)
bullColor = input.color(color.rgb(22, 230, 120), "Buy color", group = groupVisuals)
bearColor = input.color(color.rgb(242, 76, 86), "Sell color", group = groupVisuals)

groupLiq = "Зоны ликвидности (SSL / BSL)"
showLiquidity = input.bool(true, "Показывать зоны ликвидности", group = groupLiq, tooltip = "Непротестированные экстремумы за последние N баров: сверху — BSL (пул ликвидаций над хаями), снизу — SSL (пул ликвидаций под лоу). Пробитый экстремум помечается как 'сдана' и перестаёт быть целью.")
liqLookback = input.int(100, "Lookback для поиска экстремумов (бары)", minval = 20, maxval = 500, group = groupLiq)
liqMinBars = input.int(5, "Мин. возраст экстремума (бары)", minval = 1, group = groupLiq, tooltip = "Сколько баров экстремум должен продержаться нетронутым, чтобы считаться пулом ликвидности.")
liqExtend = input.int(80, "Проекция зоны вправо (бары)", minval = 10, maxval = 300, group = groupLiq)
liqMaxZones = input.int(6, "Макс. зон на сторону", minval = 1, maxval = 15, group = groupLiq)
liqBullColor = input.color(color.rgb(255, 196, 64), "BSL color (над графиком)", group = groupLiq)
liqBearColor = input.color(color.rgb(124, 168, 255), "SSL color (под графиком)", group = groupLiq)

// ---------- generic per-TF state via request.security ----------
// NOTE: security() expression must NOT contain local vars; only Pine builtins + global inputs.
// Effective values: override if > 0, else auto-by-chart-timeframe.
regmeLength    = regmeLengthIn > 0 ? regmeLengthIn : cfgRegmeLen
volMultEntry   = volMultEntryIn > 0 ? volMultEntryIn : cfgVolMult
minBodyAtr     = minBodyAtrIn > 0 ? minBodyAtrIn : cfgMinBodyAtr

[eStrBullE, eStrBearE] = request.security(syminfo.tickerid, entryTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)
[cStrBullC, cStrBearC] = request.security(syminfo.tickerid, confirmTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)
[tStrBullT, tStrBearT] = request.security(syminfo.tickerid, trendTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)

[eMomBullE, eMomBearE] = request.security(syminfo.tickerid, entryTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultEntry, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultEntry], lookahead = barmerge.lookahead_off)
[cMomBullC, cMomBearC] = request.security(syminfo.tickerid, confirmTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig], lookahead = barmerge.lookahead_off)
[tMomBullT, tMomBearT] = request.security(syminfo.tickerid, trendTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig], lookahead = barmerge.lookahead_off)

mtfBull = (eStrBullE ? 1 : 0) + (cStrBullC ? 1 : 0) + (tStrBullT ? 1 : 0)
mtfBear = (eStrBearE ? 1 : 0) + (cStrBearC ? 1 : 0) + (tStrBearT ? 1 : 0)

vol = nz(volume, 0.0)
atr = ta.atr(atrLength)
vwmaPrice = ta.vwma(hlc3, regmeLength)
regme = ta.ema(vwmaPrice, regmeSmooth)
priceAboveRegme = close >= regme
plot(regme, "REGME trend", color = priceAboveRegme ? bullColor : bearColor, linewidth = 4)

// ---------- liquidity pools (equal highs = BSL above, equal lows = SSL below) ----------
liqClusterTop(float[] arr, array<bool> sweptArr, float lvl) =>
    found = false
    n = array.size(arr)
    if n > 0
        for i = 0 to n - 1
            if not array.get(sweptArr, i) and math.abs(array.get(arr, i) - lvl) <= syminfo.mintick
                found := true
    found

var array<float> bslLevels = array.new<float>()
var array<int>   bslBirth = array.new<int>()
var array<bool>  bslSwept = array.new<bool>()
var array<line>  bslLines = array.new<line>()
var array<label> bslLabels = array.new<label>()
var array<float> sslLevels = array.new<float>()
var array<int>   sslBirth = array.new<int>()
var array<bool>  sslSwept = array.new<bool>()
var array<line>  sslLines = array.new<line>()
var array<label> sslLabels = array.new<label>()

if showLiquidity and not na(high[1]) and bar_index > liqMinBars + 1
    // Swing high/low: local extremum confirmed after liqMinBars+1 bars (no repaint: level taken from history).
    swingHi = high[liqMinBars + 1] == ta.highest(high, 2 * (liqMinBars + 1) + 1)
    swingLo = low[liqMinBars + 1] == ta.lowest(low, 2 * (liqMinBars + 1) + 1)
    eqHigh = math.sum(swingHi ? 1 : 0, liqLookback) >= 2
    eqLow = math.sum(swingLo ? 1 : 0, liqLookback) >= 2
    if swingHi and eqHigh and not liqClusterTop(bslLevels, bslSwept, high[liqMinBars + 1])
        lvl = high[liqMinBars + 1]
        line ln = line.new(bar_index - liqMinBars - 1, lvl, bar_index + liqExtend, lvl, extend = extend.right, color = color.new(liqBullColor, 45), style = line.style_dashed, width = 1)
        label lb = label.new(bar_index - liqMinBars - 1, lvl, "BSL", xloc = xloc.bar_index, yloc = yloc.price, color = color.new(liqBullColor, 85), textcolor = color.new(liqBullColor, 20), style = label.style_label_down, size = size.tiny)
        array.push(bslLevels, lvl)
        array.push(bslBirth, bar_index)
        array.push(bslSwept, false)
        array.push(bslLines, ln)
        array.push(bslLabels, lb)
        if array.size(bslLevels) > liqMaxZones
            oldLn = array.shift(bslLines)
            oldLb = array.shift(bslLabels)
            if not na(oldLn)
                line.delete(oldLn)
            if not na(oldLb)
                label.delete(oldLb)
            array.shift(bslLevels)
            array.shift(bslBirth)
            array.shift(bslSwept)
    if swingLo and eqLow and not liqClusterTop(sslLevels, sslSwept, low[liqMinBars + 1])
        lvl2 = low[liqMinBars + 1]
        line ln2 = line.new(bar_index - liqMinBars - 1, lvl2, bar_index + liqExtend, lvl2, extend = extend.right, color = color.new(liqBearColor, 45), style = line.style_dashed, width = 1)
        label lb2 = label.new(bar_index - liqMinBars - 1, lvl2, "SSL", xloc = xloc.bar_index, yloc = yloc.price, color = color.new(liqBearColor, 85), textcolor = color.new(liqBearColor, 20), style = label.style_label_up, size = size.tiny)
        array.push(sslLevels, lvl2)
        array.push(sslBirth, bar_index)
        array.push(sslSwept, false)
        array.push(sslLines, ln2)
        array.push(sslLabels, lb2)
        if array.size(sslLevels) > liqMaxZones
            oldLn2 = array.shift(sslLines)
            oldLb2 = array.shift(sslLabels)
            if not na(oldLn2)
                line.delete(oldLn2)
            if not na(oldLb2)
                label.delete(oldLb2)
            array.shift(sslLevels)
            array.shift(sslBirth)
            array.shift(sslSwept)

// Sweep detection: close beyond the pool level = liquidity taken -> mark swept, recolor.
if showLiquidity and array.size(bslLevels) > 0
    for i = 0 to array.size(bslLevels) - 1
        if not array.get(bslSwept, i) and close > array.get(bslLevels, i)
            array.set(bslSwept, i, true)
            sweptLine = array.get(bslLines, i)
            sweptLbl = array.get(bslLabels, i)
            if not na(sweptLine)
                line.set_color(sweptLine, color.new(bearColor, 60))
                line.set_style(sweptLine, line.style_solid)
            if not na(sweptLbl)
                label.set_text(sweptLbl, "BSL сдана")
                label.set_textcolor(sweptLbl, color.new(color.gray, 30))

if showLiquidity and array.size(sslLevels) > 0
    for i = 0 to array.size(sslLevels) - 1
        if not array.get(sslSwept, i) and close < array.get(sslLevels, i)
            array.set(sslSwept, i, true)
            sweptLine2 = array.get(sslLines, i)
            sweptLbl2 = array.get(sslLabels, i)
            if not na(sweptLine2)
                line.set_color(sweptLine2, color.new(bullColor, 60))
                line.set_style(sweptLine2, line.style_solid)
            if not na(sweptLbl2)
                label.set_text(sweptLbl2, "SSL сдана")
                label.set_textcolor(sweptLbl2, color.new(color.gray, 30))

// Expire stale unswept pools so the chart stays clean.
if array.size(bslLevels) > 0
    for i = array.size(bslLevels) - 1 to 0
        if bar_index - array.get(bslBirth, i) > liqLookback and not array.get(bslSwept, i)
            eLn = array.get(bslLines, i)
            eLb = array.get(bslLabels, i)
            if not na(eLn)
                line.delete(eLn)
            if not na(eLb)
                label.delete(eLb)
            array.remove(bslLines, i)
            array.remove(bslLabels, i)
            array.remove(bslLevels, i)
            array.remove(bslBirth, i)
            array.remove(bslSwept, i)

if array.size(sslLevels) > 0
    for i = array.size(sslLevels) - 1 to 0
        if bar_index - array.get(sslBirth, i) > liqLookback and not array.get(sslSwept, i)
            eLn2 = array.get(sslLines, i)
            eLb2 = array.get(sslLabels, i)
            if not na(eLn2)
                line.delete(eLn2)
            if not na(eLb2)
                label.delete(eLb2)
            array.remove(sslLines, i)
            array.remove(sslLabels, i)
            array.remove(sslLevels, i)
            array.remove(sslBirth, i)
            array.remove(sslSwept, i)

// ---------- VDO on entry TF ----------
signedVolume = close > close[1] ? vol : close < close[1] ? -vol : 0.0
totalVolume = ta.ema(vol, vdoLength)
volumePosition = ta.ema(signedVolume, vdoLength)
vzo = totalVolume != 0.0 ? 100.0 * volumePosition / totalVolume : 0.0
vdo = math.max(0.0, math.min(100.0, 50.0 + vzo * 0.5))
vdoSignalEma = ta.ema(vdo, vdoSigLen)

// ---------- session windows (yekaterinburg time, UTC+5): only two active windows give positive expectancy on GOLD ----------
// NOTE: the old timezone-based hour call was invalid in Pine v6 (CE10272); hours are now computed as UTC hour + manual offset.
srvOffsetH = input.int(5, "Часовой сдвиг от UTC (ЕКБ = 5)", minval = -12, maxval = 14, group = groupSession, tooltip = "Смещение вашего часового пояса относительно UTC. Екатеринбург (UTC+5) = 5. Если хотите видеть окна в время сервера MT5 (GMT+3), поставьте 3.")
ekbHourRaw = hour(time, "GMT") + srvOffsetH
ekbHour = ekbHourRaw >= 24 ? ekbHourRaw - 24 : ekbHourRaw < 0 ? ekbHourRaw + 24 : ekbHourRaw
inEuropeWin = sessEurope and ekbHour >= 11 and ekbHour < 15
inUsWin = sessUs and ekbHour >= 17 and ekbHour < 21
inCustomWin = sessCustomFrom >= 0 and (sessCustomFrom <= sessCustomTo ? (ekbHour >= sessCustomFrom and ekbHour < sessCustomTo + 1) : (ekbHour >= sessCustomFrom or ekbHour < sessCustomTo + 1))
inTradeWindow = inEuropeWin or inUsWin or inCustomWin
// Торговое окно показывается только цветом строки таблицы: зелёный — торгуем, красный — нет (без текста во втором столбце)
sessionRowBg = not useSessionFilter ? color.new(color.gray, 60) : color.new(inTradeWindow ? bullColor : bearColor, 75)

// ---------- status table (Russian) ----------
cornerPos = tableCorner == "Верх слева" ? position.top_left : tableCorner == "Низ слева" ? position.bottom_left : tableCorner == "Низ справа" ? position.bottom_right : position.top_right
var table panel = table.new(cornerPos, 2, 4, bgcolor = color.new(color.black, 20), border_width = 1, border_color = color.new(color.gray, 70))

tfRu(tf) =>
    tf == "D" ? "Д1" : tf == "240" ? "4ч" : tf == "60" ? "1ч" : tf == "15" ? "15м" : tf == "5" ? "5м" : tf == "3" ? "3м" : tf == "1" ? "1м" : tf

sideRu(bull, bear) => bull ? "растёт" : bear ? "падает" : "флэт"

phaseTxt = mtfBull == 3 ? "Сильный бычий тренд" : mtfBear == 3 ? "Сильный медвежий тренд" : mtfBull == 2 ? "Бычий тренд" : mtfBear == 2 ? "Медвежий тренд" : mtfBull == 1 ? "Слабо бычий (сомнительно)" : mtfBear == 1 ? "Слабо медвежий (сомнительно)" : "Флэт / разнобой"
phaseCol = mtfBull > mtfBear ? bullColor : mtfBear > mtfBull ? bearColor : color.gray

if showTable and barstate.islast
    table.cell(panel, 0, 0, "Фаза рынка", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 0, phaseTxt, text_color = phaseCol, text_size = size.small)
    table.cell(panel, 0, 1, "Тренды ТФ", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 1, tfRu(timeframe.period) + ": " + sideRu(eStrBullE, eStrBearE) + " | " + tfRu(confirmTf) + ": " + sideRu(cStrBullC, cStrBearC) + " | " + tfRu(trendTf) + ": " + sideRu(tStrBullT, tStrBearT), text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 2, "Импульс (движение)", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 2, eMomBullE or cMomBullC or tMomBullT ? "бычий импульс" : eMomBearE or cMomBearC or tMomBearT ? "медвежий импульс" : "спокойно", text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 3, "Торговое окно", text_color = color.silver, text_size = size.small, bgcolor = sessionRowBg)
    table.cell(panel, 1, 3, "", text_size = size.small, bgcolor = sessionRowBg)
