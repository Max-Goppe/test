//@version=6
// This is a Pine Script v6 INDICATOR (not a strategy) - it only displays signals, no orders.
indicator("INV REGME+VDO Signal Panel", "RVP SIGNAL", overlay = true, max_labels_count = 50, max_boxes_count = 50, max_lines_count = 50, precision = 2)

// Combined signal engine for REGME + VDO, following the tuning conclusions from the MT5 GOLD M5 export:
//  - best reaction on H1; zone-touch entries with trend; VDO band-breakout confirmation;
//  - one merged script so you don't need request.security hacks across two private indicators.
// Timeframe presets identical to regme_dynamic_zones.pine / vdo_volume_direction.pine.

groupTf = "Timeframe preset"
tfMode = input.string("Auto", "Preset", options = ["Auto", "M5", "M15", "H1"], group = groupTf,
     tooltip = "Auto = picks settings by current chart timeframe. Recommended: H1.")

bool isM5  = tfMode == "M5"  or (tfMode == "Auto" and timeframe.period == "5")
bool isM15 = tfMode == "M15" or (tfMode == "Auto" and timeframe.period == "15")
bool isH1  = tfMode == "H1"  or (tfMode == "Auto" and timeframe.period == "60")

int   cfgRegmeLength   = isM15 ? 100 : 55
float cfgVolumeMult    = isH1 ? 2.8 : isM15 ? 1.8 : 1.3
float cfgMinBodyAtr    = isH1 ? 0.50 : isM15 ? 0.50 : 0.30
float cfgMinRangeAtr   = isH1 ? 0.80 : isM15 ? 0.90 : 0.70
int   cfgVolumeBaseLen = isH1 ? 34 : 34
int   cfgVdoBandLength = isH1 ? 50 : 34
float cfgVdoBandMult   = isH1 ? 1.0 : 1.35

groupRegme = "REGME core"
regmeMode = input.string("VWMA + EMA", "Mean type", options = ["VWMA + EMA", "Session VWAP", "LinReg VWMA"], group = groupRegme)
regmeLength = input.int(0, "REGME length (0 = from preset)", minval = 0, group = groupRegme)
regmeSmooth = input.int(8, "REGME smoothing", minval = 1, group = groupRegme)
showRegmeLine = input.bool(true, "Show REGME line", group = groupRegme)

groupImpulse = "Impulse filter"
volumeLength = input.int(0, "Volume baseline (0 = from preset)", minval = 0, group = groupImpulse)
volumeMultiplier = input.float(0.0, "Volume multiplier (0 = from preset)", minval = 0.0, step = 0.05, group = groupImpulse)
atrLength = input.int(14, "ATR length", minval = 1, group = groupImpulse)
minBodyAtr = input.float(0.0, "Min body / ATR (0 = from preset)", minval = 0.0, step = 0.05, group = groupImpulse)
minRangeAtr = input.float(0.0, "Min range / ATR (0 = from preset)", minval = 0.0, step = 0.05, group = groupImpulse)

groupZones = "Dynamic zones"
showZones = input.bool(true, "Show dynamic zones", group = groupZones)
maxZones = input.int(16, "Max zones per side", minval = 1, maxval = 20, group = groupZones)
zoneExtendBars = input.int(250, "Zone projection bars", minval = 10, maxval = 500, group = groupZones)
zoneUseWicks = input.bool(true, "Use wick-to-body zones", group = groupZones)

groupVdo = "VDO oscillator"
vdoLength = input.int(10, "VDO length", minval = 2, group = groupVdo)
vdoSignalLength = input.int(13, "VDO signal length", minval = 1, group = groupVdo)
vdoBandLength = input.int(0, "VDO band length (0 = from preset)", minval = 0, group = groupVdo)
vdoBandMult = input.float(0.0, "VDO band width (0 = from preset)", minval = 0.0, step = 0.05, group = groupVdo)

groupSignal = "Signal rules"
minConfluence = input.int(2, "Min confluence score (2-3)", minval = 2, maxval = 3, group = groupSignal,
     tooltip = "Backtest GOLD H1 2022-2026: 2 works best. On M15 set 3 (strict).")
touchLookback = input.int(1, "Zone touch valid N bars", minval = 1, maxval = 10, group = groupSignal)
breakoutLookback = input.int(3, "VDO breakout valid N bars", minval = 1, maxval = 10, group = groupSignal)
impulseAgeMax = input.int(40, "Impulse zone age limit (bars)", minval = 3, maxval = 100, group = groupSignal)
requireVdo = input.bool(true, "Require VDO confirmation", group = groupSignal)
cooldownBars = input.int(20, "Signal cooldown bars", minval = 0, maxval = 100, group = groupSignal)

groupRisk = "TP / SL levels"
slMode = input.string("Zone edge + buffer", "Stop-loss basis", options = ["Zone edge + buffer", "ATR multiple", "Fixed %"], group = groupRisk)
slAtrMult = input.float(1.5, "SL ATR multiple (mode: ATR)", minval = 0.2, step = 0.1, group = groupRisk)
slPct = input.float(0.3, "SL fixed % (mode: Fixed %)", minval = 0.01, step = 0.05, group = groupRisk)
zoneBufferAtr = input.float(0.15, "Zone-edge buffer / ATR", minval = 0.0, step = 0.05, group = groupRisk)
tpMode = input.string("R multiple", "Take-profit basis", options = ["R multiple", "Opposite zone", "ATR multiple"], group = groupRisk)
rrRatio = input.float(2.0, "Reward-to-risk ratio (mode: R multiple)", minval = 0.5, step = 0.25, group = groupRisk)
tpAtrMult = input.float(3.0, "TP ATR multiple (mode: ATR)", minval = 0.5, step = 0.25, group = groupRisk)
showTpSlLines = input.bool(true, "Show TP/SL lines on chart", group = groupRisk)
maxTpSlMarkers = input.int(6, "Max open TP/SL markers", minval = 1, maxval = 20, group = groupRisk)

groupVisuals = "Visuals"
signalStyle = input.string("Flag", "Signal shape", options = ["Flag", "Arrow", "Label only"], group = groupVisuals)
showTable = input.bool(true, "Show status table", group = groupVisuals)
tableCorner = input.string("Top right", "Table position", options = ["Top left", "Top right", "Bottom left", "Bottom right"], group = groupVisuals)
bullColor = input.color(color.rgb(22, 230, 120), "Buy color", group = groupVisuals)
bearColor = input.color(color.rgb(242, 76, 86), "Sell color", group = groupVisuals)

// Effective values: manual override if > 0, otherwise the timeframe preset.
int   effRegmeLength   = regmeLength > 0 ? regmeLength : cfgRegmeLength
int   effVolBaseLen    = volumeLength > 0 ? volumeLength : cfgVolumeBaseLen
float effVolMult       = volumeMultiplier > 0 ? volumeMultiplier : cfgVolumeMult
float effMinBodyAtr    = minBodyAtr > 0 ? minBodyAtr : cfgMinBodyAtr
float effMinRangeAtr   = minRangeAtr > 0 ? minRangeAtr : cfgMinRangeAtr
int   effVdoBandLength = vdoBandLength > 0 ? vdoBandLength : cfgVdoBandLength
float effVdoBandMult   = vdoBandMult > 0 ? vdoBandMult : cfgVdoBandMult

// ────────────────────────── Core calcs (same math as both source scripts) ──────────────────────────
float vol = nz(volume, 0.0)
float atr = ta.atr(atrLength)
float volBase = ta.sma(vol, effVolBaseLen)
bool volumeImpulse = volBase > 0 and vol > volBase * effVolMult
float body = math.abs(close - open)
float candleRange = high - low
bool rangeImpulse = atr > 0 and body >= atr * effMinBodyAtr and candleRange >= atr * effMinRangeAtr

float vwmaPrice = ta.vwma(hlc3, effRegmeLength)
float regmeRaw = switch regmeMode
    "Session VWAP" => ta.vwap(hlc3)
    "LinReg VWMA" => ta.linreg(vwmaPrice, regmeSmooth, 0)
    => ta.ema(vwmaPrice, regmeSmooth)
float regme = na(regmeRaw) ? ta.ema(hlc3, regmeSmooth) : regmeRaw
bool priceAboveRegme = close >= regme

plot(showRegmeLine ? regme : na, "REGME", color = priceAboveRegme ? bullColor : bearColor, linewidth = 4)

bool bullishImpulse = showZones and volumeImpulse and rangeImpulse and close > open and close > regme
bool bearishImpulse = showZones and volumeImpulse and rangeImpulse and close < open and close < regme

var array<float> demandTops = array.new<float>()
var array<float> demandBottoms = array.new<float>()
var array<int>   demandBirth = array.new<int>()
var array<bool>  demandTouched = array.new<bool>()
var array<box>   demandBoxes = array.new<box>()
var array<float> supplyTops = array.new<float>()
var array<float> supplyBottoms = array.new<float>()
var array<int>   supplyBirth = array.new<int>()
var array<bool>  supplyTouched = array.new<bool>()
var array<box>   supplyBoxes = array.new<box>()

if bullishImpulse
    float zTop = zoneUseWicks ? math.min(open, close) : math.max(open, close)
    array.push(demandTops, zTop)
    array.push(demandBottoms, low)
    array.push(demandBirth, bar_index)
    array.push(demandTouched, false)
    box dBoxNew = showZones ? box.new(bar_index, zTop, bar_index + zoneExtendBars, low, extend = extend.right, border_color = color.new(bullColor, 45), bgcolor = color.new(bullColor, 87), border_width = 1) : na
    array.push(demandBoxes, dBoxNew)
    if array.size(demandTops) > maxZones
        box.delete(array.shift(demandBoxes))
        array.shift(demandTops)
        array.shift(demandBottoms)
        array.shift(demandBirth)
        array.shift(demandTouched)

if bearishImpulse
    float zBottom = zoneUseWicks ? math.max(open, close) : math.min(open, close)
    array.push(supplyTops, high)
    array.push(supplyBottoms, zBottom)
    array.push(supplyBirth, bar_index)
    array.push(supplyTouched, false)
    box sBoxNew = showZones ? box.new(bar_index, high, bar_index + zoneExtendBars, zBottom, extend = extend.right, border_color = color.new(bearColor, 45), bgcolor = color.new(bearColor, 87), border_width = 1) : na
    array.push(supplyBoxes, sBoxNew)
    if array.size(supplyTops) > maxZones
        box.delete(array.shift(supplyBoxes))
        array.shift(supplyTops)
        array.shift(supplyBottoms)
        array.shift(supplyBirth)
        array.shift(supplyTouched)

// First-touch tracking: boxes are created ONCE at zone birth; later bars only move the right edge / recolor.
var int lastDemandTouchBar = 0
var float lastDemandTouchTop = na
var float lastDemandTouchBottom = na
var int lastSupplyTouchBar = 0
var float lastSupplyTouchTop = na
var float lastSupplyTouchBottom = na
bool touchDemandThisBar = false
bool touchSupplyThisBar = false

if array.size(demandTops) > 0
    for i = 0 to array.size(demandTops) - 1
        float top = array.get(demandTops, i)
        float bottom = array.get(demandBottoms, i)
        int birth = array.get(demandBirth, i)
        bool wasTested = array.get(demandTouched, i)
        bool broken = close < bottom
        bool inZone = low <= top and high >= bottom
        bool tested = broken or wasTested ? wasTested : inZone and bar_index > birth
        if tested and not wasTested
            lastDemandTouchBar := bar_index
            lastDemandTouchTop := top
            lastDemandTouchBottom := bottom
            touchDemandThisBar := true
        if tested
            array.set(demandTouched, i, true)
        box dB = array.get(demandBoxes, i)
        if showZones and not na(dB)
            if broken
                box.delete(dB)
                array.set(demandBoxes, i, na)
            else
                box.set_right(dB, bar_index + zoneExtendBars)
                if tested
                    box.set_bgcolor(dB, color.new(bullColor, 79))
                    box.set_border_color(dB, color.new(bullColor, 15))

if array.size(supplyTops) > 0
    for i = 0 to array.size(supplyTops) - 1
        float top = array.get(supplyTops, i)
        float bottom = array.get(supplyBottoms, i)
        int birth = array.get(supplyBirth, i)
        bool wasTested = array.get(supplyTouched, i)
        bool broken = close > top
        bool inZone = high >= bottom and low <= top
        bool tested = broken or wasTested ? wasTested : inZone and bar_index > birth
        if tested and not wasTested
            lastSupplyTouchBar := bar_index
            lastSupplyTouchTop := top
            lastSupplyTouchBottom := bottom
            touchSupplyThisBar := true
        if tested
            array.set(supplyTouched, i, true)
        box sB = array.get(supplyBoxes, i)
        if showZones and not na(sB)
            if broken
                box.delete(sB)
                array.set(supplyBoxes, i, na)
            else
                box.set_right(sB, bar_index + zoneExtendBars)
                if tested
                    box.set_bgcolor(sB, color.new(bearColor, 79))
                    box.set_border_color(sB, color.new(bearColor, 15))

// Fresh first-touch of an opposite-side zone within the lookback window.
bool freshDemandTouch = lastDemandTouchBar > 0 and bar_index - lastDemandTouchBar <= touchLookback and priceAboveRegme
bool freshSupplyTouch = lastSupplyTouchBar > 0 and bar_index - lastSupplyTouchBar <= touchLookback and not priceAboveRegme

// Impulse freshness: a qualifying impulse happened recently and we are still on that side of REGME.
// ta.barssince() must be called at global scope (not inside table blocks later).
bool bullImpulseNow = bullishImpulse
bool bearImpulseNow = bearishImpulse
int bullImpulseAge = ta.barssince(bullImpulseNow)
int bearImpulseAge = ta.barssince(bearImpulseNow)
bool freshBullImpulse = not na(bullImpulseAge) and bullImpulseAge <= impulseAgeMax and priceAboveRegme
bool freshBearImpulse = not na(bearImpulseAge) and bearImpulseAge <= impulseAgeMax and not priceAboveRegme

// VDO oscillator + dynamic bands (mirrors vdo_volume_direction.pine).
float signedVolume = close > close[1] ? vol : close < close[1] ? -vol : 0.0
float totalVolume = ta.ema(vol, vdoLength)
float volumePosition = ta.ema(signedVolume, vdoLength)
float vzo = totalVolume != 0.0 ? 100.0 * volumePosition / totalVolume : 0.0
float vdo = math.max(0.0, math.min(100.0, 50.0 + vzo * 0.5))
float vdoSignalEma = ta.ema(vdo, vdoSignalLength)
float vdoDev = ta.stdev(vdo, effVdoBandLength)
float upperBand = math.min(80.0, vdoSignalEma + vdoDev * effVdoBandMult)
float lowerBand = math.max(20.0, vdoSignalEma - vdoDev * effVdoBandMult)

int vdoBullBars = ta.barssince(ta.crossover(vdo, upperBand))
int vdoBearBars = ta.barssince(ta.crossunder(vdo, lowerBand))
bool vdoBullBreakout = not na(vdoBullBars) and vdoBullBars <= breakoutLookback and vdo > vdoSignalEma
bool vdoBearBreakout = not na(vdoBearBars) and vdoBearBars <= breakoutLookback and vdo < vdoSignalEma

// ────────────────────────── Confluence scoring ──────────────────────────
int bullScore = (freshBullImpulse ? 1 : 0) + (freshDemandTouch ? 1 : 0) + (not requireVdo or vdoBullBreakout ? 1 : 0)
int bearScore = (freshBearImpulse ? 1 : 0) + (freshSupplyTouch ? 1 : 0) + (not requireVdo or vdoBearBreakout ? 1 : 0)

var int lastSignalBar = 0
bool inCooldown = lastSignalBar > 0 and bar_index - lastSignalBar <= cooldownBars

bool rawBuy = priceAboveRegme and bullScore >= minConfluence
bool rawSell = not priceAboveRegme and bearScore >= minConfluence

bool buySignal = rawBuy and not inCooldown
bool sellSignal = rawSell and not inCooldown

if buySignal or sellSignal
    lastSignalBar := bar_index

// ────────────────────────── TP / SL levels ──────────────────────────
// Entry = signal-bar close. SL basis: touched-zone edge + buffer (fallback ATR), ATR multiple, or fixed %.
// TP basis: R multiple of the risk (default RR 2.0), opposite zone edge, or ATR multiple.
float slDistAtr = atr * slAtrMult
float slDistPct = close * slPct / 100.0
float buyZoneEdgeSl = not na(lastDemandTouchBottom) and bar_index - lastDemandTouchBar <= touchLookback ? close - (lastDemandTouchBottom - atr * zoneBufferAtr) : na
float sellZoneEdgeSl = not na(lastSupplyTouchTop) and bar_index - lastSupplyTouchBar <= touchLookback ? (lastSupplyTouchTop + atr * zoneBufferAtr) - close : na
float rawBuySl = slMode == "ATR multiple" ? slDistAtr : slMode == "Fixed %" ? slDistPct : nz(buyZoneEdgeSl, slDistAtr)
float rawSellSl = slMode == "ATR multiple" ? slDistAtr : slMode == "Fixed %" ? slDistPct : nz(sellZoneEdgeSl, slDistAtr)
float buySlDist = math.max(0.05 * atr, rawBuySl)
float sellSlDist = math.max(0.05 * atr, rawSellSl)

// Nearest untouched supply/demand zone edge for the "Opposite zone" TP mode.
float nearestSupplyTop = na
if array.size(supplyTops) > 0
    for i = 0 to array.size(supplyTops) - 1
        if not array.get(supplyTouched, i)
            float t = array.get(supplyTops, i)
            if t > close and (na(nearestSupplyTop) or t < nearestSupplyTop)
                nearestSupplyTop := t
float nearestDemandBottom = na
if array.size(demandTops) > 0
    for i = 0 to array.size(demandTops) - 1
        if not array.get(demandTouched, i)
            float b = array.get(demandBottoms, i)
            if b < close and (na(nearestDemandBottom) or b > nearestDemandBottom)
                nearestDemandBottom := b

float rawBuyTp = tpMode == "R multiple" ? buySlDist * rrRatio : tpMode == "ATR multiple" ? atr * tpAtrMult : not na(nearestSupplyTop) ? math.max(atr * 0.5, nearestSupplyTop - close) : buySlDist * rrRatio
float rawSellTp = tpMode == "R multiple" ? sellSlDist * rrRatio : tpMode == "ATR multiple" ? atr * tpAtrMult : not na(nearestDemandBottom) ? math.max(atr * 0.5, close - nearestDemandBottom) : sellSlDist * rrRatio
float buyTpDist = math.max(0.1 * atr, rawBuyTp)
float sellTpDist = math.max(0.1 * atr, rawSellTp)

var float curEntry = na
var float curSl = na
var float curTp = na
var int   curDir = 0
var int   curSigBar = 0
var line  tpLine = na
var line  slLine = na
var label tpLbl = na
var label slLbl = na

// Open a fresh TP/SL marker on each new signal (replaces the previous one).
if buySignal or sellSignal
    curEntry := close
    curDir := buySignal ? 1 : -1
    curSl := buySignal ? close - buySlDist : close + sellSlDist
    curTp := buySignal ? close + buyTpDist : close - sellTpDist
    curSigBar := bar_index
    line.delete(tpLine)
    line.delete(slLine)
    label.delete(tpLbl)
    label.delete(slLbl)
    if showTpSlLines and not na(curEntry)
        tpLine := line.new(bar_index, curTp, bar_index + 5, curTp, extend = extend.right, color = color.new(bullColor, 20), style = line.style_solid, width = 2)
        slLine := line.new(bar_index, curSl, bar_index + 5, curSl, extend = extend.right, color = color.new(bearColor, 20), style = line.style_dashed, width = 2)
        tpLbl := label.new(bar_index, curTp, "TP " + str.tostring(curTp, format.mintick), style = label.style_none, color = color.new(color.white, 100), textcolor = bullColor, size = size.small)
        slLbl := label.new(bar_index, curSl, "SL " + str.tostring(curSl, format.mintick), style = label.style_none, color = color.new(color.white, 100), textcolor = bearColor, size = size.small)

// Manage the open position: close on TP/SL hit or on an opposite signal; then fade the markers out.
bool posOpen = not na(curEntry)
string exitReason = ""
if posOpen
    bool tpHit = curDir == 1 ? high >= curTp : low <= curTp
    bool slHit = curDir == 1 ? low <= curSl : high >= curSl
    bool flip = curDir == 1 ? sellSignal : buySignal
    if tpHit
        exitReason := "TP"
    else if slHit
        exitReason := "SL"
    else if flip
        exitReason := "FLIP"
    if exitReason != ""
        curEntry := na
        curSl := na
        curTp := na
        curDir := 0
    else if showTpSlLines
        if not na(tpLine)
            line.set_right(tpLine, bar_index + 5)
        if not na(slLine)
            line.set_right(slLine, bar_index + 5)

// After exit, keep the lines visible for a few bars, then delete them.
var int exitFadeBar = 0
if exitReason != ""
    exitFadeBar := bar_index
if showTpSlLines and exitReason == "" and exitFadeBar > 0 and bar_index - exitFadeBar > 8
    line.delete(tpLine)
    line.delete(slLine)
    label.delete(tpLbl)
    label.delete(slLbl)
    tpLine := na
    slLine := na
    tpLbl := na
    slLbl := na
    exitFadeBar := 0

// ────────────────────────── Visuals ──────────────────────────
// Signals and zone-touch marks drawn via label.new() only: label arguments accept
// series values, so no const-arg compile errors (CE10123/CE10272) are possible here.
if buySignal
    string buyText = signalStyle == "Arrow" ? "↑ BUY" : "⚑ BUY"
    label.new(bar_index, low, buyText, style = label.style_label_up, color = bullColor, textcolor = color.black, size = size.normal)
if sellSignal
    string sellText = signalStyle == "Arrow" ? "↓ SELL" : "⚑ SELL"
    label.new(bar_index, high, sellText, style = label.style_label_down, color = bearColor, textcolor = color.black, size = size.normal)
if signalStyle == "Label only" and buySignal
    label.new(bar_index, low, "BUY", style = label.style_none, color = color.new(color.white, 100), textcolor = bullColor, size = size.normal)
if signalStyle == "Label only" and sellSignal
    label.new(bar_index, high, "SELL", style = label.style_none, color = color.new(color.white, 100), textcolor = bearColor, size = size.normal)

if touchDemandThisBar
    label.new(bar_index, low, "T", style = label.style_label_up, color = color.new(bullColor, 60), textcolor = bullColor, size = size.tiny)
if touchSupplyThisBar
    label.new(bar_index, high, "T", style = label.style_label_down, color = color.new(bearColor, 60), textcolor = bearColor, size = size.tiny)

// ────────────────────────── Status table ──────────────────────────
cornerPos = tableCorner == "Top left" ? position.top_left : tableCorner == "Bottom left" ? position.bottom_left : tableCorner == "Bottom right" ? position.bottom_right : position.top_right
var table panel = table.new(cornerPos, 2, 6, bgcolor = color.new(color.black, 20), border_width = 1, border_color = color.new(color.gray, 70))

string stateTxt = buySignal ? "BUY" : sellSignal ? "SELL" : priceAboveRegme ? "Bullish" : "Bearish"
color stateCol = buySignal ? bullColor : sellSignal ? bearColor : priceAboveRegme ? color.new(bullColor, 40) : color.new(bearColor, 40)

if showTable and barstate.islast
    table.cell(panel, 0, 0, "STATE", text_color = stateCol, text_size = size.large)
    table.cell(panel, 1, 0, stateTxt, text_color = stateCol, text_size = size.large)
    table.cell(panel, 0, 1, "REGME side", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 1, priceAboveRegme ? "above (bull)" : "below (bear)", text_color = priceAboveRegme ? bullColor : bearColor, text_size = size.small)
    table.cell(panel, 0, 2, "Fresh impulse", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 2, freshBullImpulse ? "bull #" + str.tostring(bullImpulseAge) : freshBearImpulse ? "bear #" + str.tostring(bearImpulseAge) : "none", text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 3, "Zone touch", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 3, freshDemandTouch ? "demand T" : freshSupplyTouch ? "supply T" : "none", text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 4, "VDO", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 4, str.tostring(vdo, "#.#") + (vdoBullBreakout ? "  ^break" : vdoBearBreakout ? "  vbreak" : ""), text_color = vdo >= vdoSignalEma ? bullColor : bearColor, text_size = size.small)
    string posTxt = posOpen ? (curDir == 1 ? "LONG" : "SHORT") + "  TP " + str.tostring(curTp, format.mintick) + "  SL " + str.tostring(curSl, format.mintick) : exitReason != "" ? "closed: " + exitReason : "flat"
    color posCol = posOpen ? color.white : exitReason == "TP" ? bullColor : exitReason == "SL" ? bearColor : color.gray
    table.cell(panel, 0, 5, "Position", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 5, posTxt, text_color = posCol, text_size = size.small)

// ────────────────────────── Alerts ──────────────────────────
alertcondition(buySignal, "RVP BUY", "BUY signal {{ticker}} {{interval}} price {{close}}")
alertcondition(sellSignal, "RVP SELL", "SELL signal {{ticker}} {{interval}} price {{close}}")
alertcondition(buySignal or sellSignal, "RVP BUY or SELL", "Signal on {{ticker}} {{interval}}")
