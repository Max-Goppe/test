//@version=6
// RVP SCALP - multitimeframe scalping indicator for XAUUSD (tuned on MT5 GOLD M5 export 2022-2026).
// Trend line + MTF confluence table + scalp BUY/SELL signals with TP/SL lines.
indicator("INV RVP SCALP", "RVP SCALP", overlay = true, max_labels_count = 50, max_boxes_count = 50, precision = 2)

groupTf = "Timeframes"
entryTf   = input.timeframe("", "Entry timeframe (empty = chart)", group = groupTf)
confirmTf = input.timeframe("15", "Confirm timeframe", group = groupTf)
trendTf   = input.timeframe("60", "Trend timeframe", group = groupTf)

groupCore = "REGME core (per TF)"
regmeLength   = input.int(55, "REGME length", minval = 5, group = groupCore)
regmeSmooth   = input.int(8, "REGME smoothing", minval = 1, group = groupCore)
atrLength     = input.int(14, "ATR length", minval = 1, group = groupCore)
volumeBaseLen = input.int(50, "Volume baseline", minval = 5, group = groupCore)
volMultEntry  = input.float(1.6, "Vol multiplier (entry TF)", minval = 0.5, step = 0.1, group = groupCore)
volMultBig    = input.float(2.0, "Vol multiplier (confirm/trend TF)", minval = 0.5, step = 0.1, group = groupCore)
minBodyAtr    = input.float(0.55, "Min body / ATR", minval = 0.0, step = 0.05, group = groupCore)

groupZones = "Zones (entry TF)"
showZones = input.bool(true, "Show dynamic zones", group = groupZones)
maxZones = input.int(16, "Max zones per side", minval = 4, maxval = 20, group = groupZones)
zoneExtendBars = input.int(120, "Zone projection bars", minval = 10, maxval = 300, group = groupZones)
touchLookback = input.int(3, "Touch valid N bars", minval = 1, maxval = 10, group = groupZones)

groupSignal = "Scalp signal rules"
requireAllTf = input.bool(true, "Require ALL timeframes aligned", group = groupSignal, tooltip = "Off = allow 2 of 3 TFs to agree.")
vdoLength = input.int(14, "VDO length", minval = 2, group = groupSignal)
vdoSigLen = input.int(8, "VDO signal length", minval = 1, group = groupSignal)
vdoBandLen = input.int(50, "VDO band length", minval = 5, group = groupSignal)
vdoBandMult = input.float(1.35, "VDO band width", minval = 0.5, step = 0.05, group = groupSignal)
breakoutLookback = input.int(3, "VDO breakout valid N bars", minval = 1, maxval = 10, group = groupSignal)
cooldownBars = input.int(6, "Signal cooldown bars", minval = 1, maxval = 30, group = groupSignal)

groupTpSl = "TP / SL levels"
slMode = input.string("Zone edge + buffer", "Stop-loss basis", options = ["Zone edge + buffer", "ATR multiple", "Fixed %"], group = groupTpSl)
slAtrMult = input.float(1.5, "SL ATR multiple", minval = 0.3, step = 0.1, group = groupTpSl)
slPct = input.float(0.3, "SL fixed %", minval = 0.05, step = 0.05, group = groupTpSl)
slBufferAtr = input.float(0.15, "Zone-edge buffer (ATR)", minval = 0.0, step = 0.05, group = groupTpSl)
tpMode = input.string("R multiple", "Take-profit basis", options = ["R multiple", "ATR multiple"], group = groupTpSl)
tpRR = input.float(2.0, "TP R-multiple", minval = 0.5, step = 0.25, group = groupTpSl)
tpAtrMult = input.float(3.0, "TP ATR multiple", minval = 0.5, step = 0.25, group = groupTpSl)
showTpSl = input.bool(true, "Show TP/SL lines", group = groupTpSl)

groupVisuals = "Visuals"
showTable = input.bool(true, "Show status table", group = groupVisuals)
tableCorner = input.string("Top right", "Table position", options = ["Top left", "Top right", "Bottom left", "Bottom right"], group = groupVisuals)
bullColor = input.color(color.rgb(22, 230, 120), "Buy color", group = groupVisuals)
bearColor = input.color(color.rgb(242, 76, 86), "Sell color", group = groupVisuals)

// ---------- generic per-TF state via request.security ----------
// NOTE: security() expression must NOT contain local vars; only Pine builtins + global inputs.
[eStrBullE, eStrBearE] = request.security(syminfo.tickerid, entryTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)
[cStrBullC, cStrBearC] = request.security(syminfo.tickerid, confirmTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)
[tStrBullT, tStrBearT] = request.security(syminfo.tickerid, trendTf, [close > ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth), close < ta.ema(ta.vwma(hlc3, regmeLength), regmeSmooth)], lookahead = barmerge.lookahead_off)

[eMomBullE, eMomBearE] = request.security(syminfo.tickerid, entryTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultEntry, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultEntry], lookahead = barmerge.lookahead_off)
[cMomBullC, cMomBearC] = request.security(syminfo.tickerid, confirmTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig], lookahead = barmerge.lookahead_off)
[tMomBullT, tMomBearT] = request.security(syminfo.tickerid, trendTf, [math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close > open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig, math.abs(close - open) >= ta.atr(atrLength) * minBodyAtr and close < open and volume > ta.sma(nz(volume, 0.0), volumeBaseLen) * volMultBig], lookahead = barmerge.lookahead_off)

int mtfBull = (eStrBullE ? 1 : 0) + (cStrBullC ? 1 : 0) + (tStrBullT ? 1 : 0)
int mtfBear = (eStrBearE ? 1 : 0) + (cStrBearC ? 1 : 0) + (tStrBearT ? 1 : 0)
bool bullAligned = requireAllTf ? mtfBull == 3 : mtfBull >= 2
bool bearAligned = requireAllTf ? mtfBear == 3 : mtfBear >= 2

// ---------- entry-TF native calcs (line, zones, VDO) ----------
float vol = nz(volume, 0.0)
float atr = ta.atr(atrLength)
float vwmaPrice = ta.vwma(hlc3, regmeLength)
float regme = ta.ema(vwmaPrice, regmeSmooth)
bool priceAboveRegme = close >= regme
plot(regme, "REGME trend", color = priceAboveRegme ? bullColor : bearColor, linewidth = 4)

float volBase = ta.sma(vol, volumeBaseLen)
bool volumeImpulse = volBase > 0 and vol > volBase * volMultEntry
bool rangeImpulse = atr > 0 and math.abs(close - open) >= atr * minBodyAtr and (high - low) >= atr * 0.9

bool bullishImpulse = showZones and volumeImpulse and rangeImpulse and close > open and close > regme
bool bearishImpulse = showZones and volumeImpulse and rangeImpulse and close < open and close < regme

var array<float> demandTops = array.new<float>()
var array<float> demandBottoms = array.new<float>()
var array<int>   demandBirth = array.new<int>()
var array<bool>  demandTouched = array.new<bool>()
var array<box>   demandBoxes = array.new<box>()
var array<float> supplyTops = array.new<float>()
var array<float> supplyBottoms = array.new<float>()
var array<int>   supplyBirth = array.new<int>()
var array<bool>  supplyTouched = array.new<bool>()
var array<box>   supplyBoxes = array.new<box>()

if bullishImpulse
    float zTop = math.min(open, close)
    array.push(demandTops, zTop)
    array.push(demandBottoms, low)
    array.push(demandBirth, bar_index)
    array.push(demandTouched, false)
    box dBoxNew = showZones ? box.new(bar_index, zTop, bar_index + zoneExtendBars, low, extend = extend.right, border_color = color.new(bullColor, 45), bgcolor = color.new(bullColor, 87), border_width = 1) : na
    array.push(demandBoxes, dBoxNew)
    if array.size(demandTops) > maxZones
        box.delete(array.shift(demandBoxes))
        array.shift(demandTops)
        array.shift(demandBottoms)
        array.shift(demandBirth)
        array.shift(demandTouched)

if bearishImpulse
    float zBottom = math.max(open, close)
    array.push(supplyTops, high)
    array.push(supplyBottoms, zBottom)
    array.push(supplyBirth, bar_index)
    array.push(supplyTouched, false)
    box sBoxNew = showZones ? box.new(bar_index, high, bar_index + zoneExtendBars, zBottom, extend = extend.right, border_color = color.new(bearColor, 45), bgcolor = color.new(bearColor, 87), border_width = 1) : na
    array.push(supplyBoxes, sBoxNew)
    if array.size(supplyTops) > maxZones
        box.delete(array.shift(supplyBoxes))
        array.shift(supplyTops)
        array.shift(supplyBottoms)
        array.shift(supplyBirth)
        array.shift(supplyTouched)

var int lastDemandTouchBar = 0
var int lastSupplyTouchBar = 0
var float lastDemandEdge = 0.0
var float lastSupplyEdge = 0.0
bool touchDemandThisBar = false
bool touchSupplyThisBar = false

if array.size(demandTops) > 0
    for i = 0 to array.size(demandTops) - 1
        float top = array.get(demandTops, i)
        float bottom = array.get(demandBottoms, i)
        int birth = array.get(demandBirth, i)
        bool wasTested = array.get(demandTouched, i)
        bool broken = close < bottom
        bool inZone = low <= top and high >= bottom
        bool tested = broken or wasTested ? wasTested : inZone and bar_index > birth
        if tested and not wasTested
            lastDemandTouchBar := bar_index
            lastDemandEdge := bottom
            touchDemandThisBar := true
        if tested
            array.set(demandTouched, i, true)
        box dB = array.get(demandBoxes, i)
        if showZones and not na(dB)
            if broken
                box.delete(dB)
                array.set(demandBoxes, i, na)
            else if tested
                box.set_bgcolor(dB, color.new(bullColor, 79))
                box.set_border_color(dB, color.new(bullColor, 15))

if array.size(supplyTops) > 0
    for i = 0 to array.size(supplyTops) - 1
        float top = array.get(supplyTops, i)
        float bottom = array.get(supplyBottoms, i)
        int birth = array.get(supplyBirth, i)
        bool wasTested = array.get(supplyTouched, i)
        bool broken = close > top
        bool inZone = high >= bottom and low <= top
        bool tested = broken or wasTested ? wasTested : inZone and bar_index > birth
        if tested and not wasTested
            lastSupplyTouchBar := bar_index
            lastSupplyEdge := top
            touchSupplyThisBar := true
        if tested
            array.set(supplyTouched, i, true)
        box sB = array.get(supplyBoxes, i)
        if showZones and not na(sB)
            if broken
                box.delete(sB)
                array.set(supplyBoxes, i, na)
            else if tested
                box.set_bgcolor(sB, color.new(bearColor, 79))
                box.set_border_color(sB, color.new(bearColor, 15))

bool freshDemandTouch = lastDemandTouchBar > 0 and bar_index - lastDemandTouchBar <= touchLookback
bool freshSupplyTouch = lastSupplyTouchBar > 0 and bar_index - lastSupplyTouchBar <= touchLookback

// ---------- VDO on entry TF ----------
float signedVolume = close > close[1] ? vol : close < close[1] ? -vol : 0.0
float totalVolume = ta.ema(vol, vdoLength)
float volumePosition = ta.ema(signedVolume, vdoLength)
float vzo = totalVolume != 0.0 ? 100.0 * volumePosition / totalVolume : 0.0
float vdo = math.max(0.0, math.min(100.0, 50.0 + vzo * 0.5))
float vdoSignalEma = ta.ema(vdo, vdoSigLen)
float vdoDev = ta.stdev(vdo, vdoBandLen)
float upperBand = math.min(80.0, vdoSignalEma + vdoDev * vdoBandMult)
float lowerBand = math.max(20.0, vdoSignalEma - vdoDev * vdoBandMult)

int vdoBullBars = ta.barssince(ta.crossover(vdo, upperBand))
int vdoBearBars = ta.barssince(ta.crossunder(vdo, lowerBand))
bool vdoBullBreakout = not na(vdoBullBars) and vdoBullBars <= breakoutLookback and vdo > vdoSignalEma
bool vdoBearBreakout = not na(vdoBearBars) and vdoBearBars <= breakoutLookback and vdo < vdoSignalEma
// pullback trigger: VDO dips into oversold / overbought extremes while the trend is up/down
bool vdoOversold = vdo < 30
bool vdoOverbought = vdo > 70

// ---------- scalp signals: all-TF alignment + (fresh zone touch OR VDO extreme/breakout) ----------
bool buySetup  = bullAligned and priceAboveRegme and (freshDemandTouch or vdoOversold or vdoBullBreakout)
bool sellSetup = bearAligned and not priceAboveRegme and (freshSupplyTouch or vdoOverbought or vdoBearBreakout)

var int lastSignalBar = 0
bool inCooldown = lastSignalBar > 0 and bar_index - lastSignalBar <= cooldownBars
bool buySignal = buySetup and not inCooldown
bool sellSignal = sellSetup and not inCooldown
if buySignal or sellSignal
    lastSignalBar := bar_index

// ---------- TP / SL ----------
var float posSide = 0.0
var float posTp = 0.0
var float posSl = 0.0
var int posClosedBar = 0
var string posResult = ""
var line tpLine = na
var line slLine = na
var label tpLbl = na
var label slLbl = na

f_stopLevel(bool _isBuy, float _price, float _edge, bool _hasEdge) =>
    float slp = _isBuy ? _price - atr * slAtrMult : _price + atr * slAtrMult
    if slMode == "Zone edge + buffer"
        if _hasEdge and _edge > 0
            slp := _isBuy ? _edge - atr * slBufferAtr : _edge + atr * slBufferAtr
    else if slMode == "Fixed %"
        slp := _isBuy ? _price * (1 - slPct / 100) : _price * (1 + slPct / 100)
    slp

if buySignal and posSide <= 0
    posSide := 1
    posSl := f_stopLevel(true, close, lastDemandEdge, lastDemandTouchBar > 0)
    float risk = math.abs(close - posSl)
    posTp := tpMode == "R multiple" ? close + risk * tpRR : close + atr * tpAtrMult
    posResult := ""
    line.delete(tpLine)
    line.delete(slLine)
    label.delete(tpLbl)
    label.delete(slLbl)
    if showTpSl
        tpLine := line.new(bar_index, posTp, bar_index + 1, posTp, extend = extend.right, color = bullColor, width = 2)
        slLine := line.new(bar_index, posSl, bar_index + 1, posSl, extend = extend.right, color = bearColor, width = 2, style = line.style_dashed)
        tpLbl := label.new(bar_index, posTp, "TP " + str.tostring(posTp, format.mintick), style = label.style_none, textcolor = bullColor, size = size.small)
        slLbl := label.new(bar_index, posSl, "SL " + str.tostring(posSl, format.mintick), style = label.style_none, textcolor = bearColor, size = size.small)

if sellSignal and posSide >= 0
    posSide := -1
    posSl := f_stopLevel(false, close, lastSupplyEdge, lastSupplyTouchBar > 0)
    float risk = math.abs(posSl - close)
    posTp := tpMode == "R multiple" ? close - risk * tpRR : close - atr * tpAtrMult
    posResult := ""
    line.delete(tpLine)
    line.delete(slLine)
    label.delete(tpLbl)
    label.delete(slLbl)
    if showTpSl
        tpLine := line.new(bar_index, posTp, bar_index + 1, posTp, extend = extend.right, color = bearColor, width = 2)
        slLine := line.new(bar_index, posSl, bar_index + 1, posSl, extend = extend.right, color = bullColor, width = 2, style = line.style_dashed)
        tpLbl := label.new(bar_index, posTp, "TP " + str.tostring(posTp, format.mintick), style = label.style_none, textcolor = bearColor, size = size.small)
        slLbl := label.new(bar_index, posSl, "SL " + str.tostring(posSl, format.mintick), style = label.style_none, textcolor = bullColor, size = size.small)

if posSide > 0 and posResult == ""
    if low <= posSl
        posResult := "closed: SL"
        posClosedBar := bar_index
    else if high >= posTp
        posResult := "closed: TP"
        posClosedBar := bar_index
    else if sellSignal
        posResult := "closed: FLIP"
        posClosedBar := bar_index
if posSide < 0 and posResult == ""
    if high >= posSl
        posResult := "closed: SL"
        posClosedBar := bar_index
    else if low <= posTp
        posResult := "closed: TP"
        posClosedBar := bar_index
    else if buySignal
        posResult := "closed: FLIP"
        posClosedBar := bar_index

if posResult != "" and bar_index - posClosedBar > 8
    line.delete(tpLine)
    line.delete(slLine)
    label.delete(tpLbl)
    label.delete(slLbl)
    tpLine := na
    slLine := na
    tpLbl := na
    slLbl := na
    posSide := 0

// ---------- signal labels ----------
if buySignal
    label.new(bar_index, low, "⚑ BUY", style = label.style_label_up, color = bullColor, textcolor = color.black, size = size.normal)
if sellSignal
    label.new(bar_index, high, "⚑ SELL", style = label.style_label_down, color = bearColor, textcolor = color.black, size = size.normal)
if touchDemandThisBar
    label.new(bar_index, low, "T", style = label.style_label_up, color = color.new(bullColor, 60), textcolor = bullColor, size = size.tiny)
if touchSupplyThisBar
    label.new(bar_index, high, "T", style = label.style_label_down, color = color.new(bearColor, 60), textcolor = bearColor, size = size.tiny)

// ---------- status table ----------
cornerPos = tableCorner == "Top left" ? position.top_left : tableCorner == "Bottom left" ? position.bottom_left : tableCorner == "Bottom right" ? position.bottom_right : position.top_right
var table panel = table.new(cornerPos, 2, 7, bgcolor = color.new(color.black, 20), border_width = 1, border_color = color.new(color.gray, 70))

string stateTxt = buySignal ? "BUY" : sellSignal ? "SELL" : bullAligned ? "Bullish" : bearAligned ? "Bearish" : "Mixed"
color stateCol = buySignal ? bullColor : sellSignal ? bearColor : bullAligned ? color.new(bullColor, 40) : bearAligned ? color.new(bearColor, 40) : color.new(color.gray, 30)

string posTxt = posSide > 0 and posResult == "" ? "LONG TP " + str.tostring(posTp, format.mintick) + " / SL " + str.tostring(posSl, format.mintick) : posSide < 0 and posResult == "" ? "SHORT TP " + str.tostring(posTp, format.mintick) + " / SL " + str.tostring(posSl, format.mintick) : posResult != "" ? posResult : "none"
color posCol = posSide > 0 ? bullColor : posSide < 0 ? bearColor : color.gray

if showTable and barstate.islast
    table.cell(panel, 0, 0, "STATE", text_color = stateCol, text_size = size.large)
    table.cell(panel, 1, 0, stateTxt, text_color = stateCol, text_size = size.large)
    table.cell(panel, 0, 1, "Trend MTF", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 1, str.tostring(mtfBull) + "▲ / " + str.tostring(mtfBear) + "▼", text_color = mtfBull > mtfBear ? bullColor : mtfBear > mtfBull ? bearColor : color.gray, text_size = size.small)
    table.cell(panel, 0, 2, "Structure (" + entryTf + "/" + confirmTf + "/" + trendTf + ")", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 2, (eStrBullE ? "▲" : eStrBearE ? "▼" : "–") + (cStrBullC ? "▲" : cStrBearC ? "▼" : "–") + (tStrBullT ? "▲" : tStrBearT ? "▼" : "–"), text_color = priceAboveRegme ? bullColor : bearColor, text_size = size.small)
    table.cell(panel, 0, 3, "Momentum", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 3, eMomBullE or cMomBullC or tMomBullT ? "bull impulse" : eMomBearE or cMomBearC or tMomBearT ? "bear impulse" : "quiet", text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 4, "Volume (entry TF)", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 4, volumeImpulse ? "impulse x" + str.tostring(vol / volBase, "#.#") : "normal", text_color = volumeImpulse ? color.yellow : color.gray, text_size = size.small)
    table.cell(panel, 0, 5, "Zone / VDO", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 5, (freshDemandTouch ? "demand T" : freshSupplyTouch ? "supply T" : "no touch") + " | VDO " + str.tostring(vdo, "#.#") + (vdoBullBreakout ? " ^brk" : vdoBearBreakout ? " vbrk" : ""), text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 6, "Position", text_color = color.silver, text_size = size.small)
    table.cell(panel, 1, 6, posTxt, text_color = posCol, text_size = size.small)

alertcondition(buySignal, "SCALP BUY", "Scalp BUY {{ticker}} {{interval}} price {{close}}")
alertcondition(sellSignal, "SCALP SELL", "Scalp SELL {{ticker}} {{interval}} price {{close}}")
